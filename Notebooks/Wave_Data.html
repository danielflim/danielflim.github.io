import pandas as pd

def merge_wave_buoy_data(file_a: str, file_b: str, output_file: str):
    # Read both CSV files
    # Keeping station_id as string prevents leading zeros (like '098') from being dropped
    df_a = pd.read_csv(file_a, dtype={"station_id": str})
    df_b = pd.read_csv(file_b, dtype={"station_id": str})

    # Ensure time strings are consistently formatted/parsed before joining
    df_a["time"] = pd.to_datetime(df_a["time"])
    df_b["time"] = pd.to_datetime(df_b["time"])

    # Merge on 'time' using an inner join (records present in both files)
    merged_df = pd.merge(df_a, df_b, on="time", suffixes=("1", "2"), how="inner")

    # Arrange columns: place 'time' first, followed by station 1 fields, then station 2 fields
    non_time_cols = [c for c in df_a.columns if c != "time"]
    ordered_cols = (
        ["time"]
        + [f"{col}1" for col in non_time_cols]
        + [f"{col}2" for col in non_time_cols]
    )

    merged_df = merged_df[ordered_cols]

    # Save to CSV
    merged_df.to_csv(output_file, index=False)
    print(f"Successfully wrote {len(merged_df)} merged records to {output_file}")

    return merged_df

import matplotlib.dates as mdates
import matplotlib.pyplot as plt
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

# 1. Prepare and clean data
data = df.sort_values("time").dropna(subset=["waveHs1", "waveHs2"]).copy()

# Features must be a 2D array / DataFrame; target is a 1D Series
X = data[["waveHs2"]]
y = data["waveHs1"]

# 2. Chronological Train/Test split (e.g., 80% train, 20% test)
split_idx = int(len(data) * 0.8)

X_train, X_test = X.iloc[:split_idx], X.iloc[split_idx:]
y_train, y_test = y.iloc[:split_idx], y.iloc[split_idx:]
test_times = data["time"].iloc[split_idx:]

# 3. Fit the OLS Linear Regression model
model = LinearRegression()
model.fit(X_train, y_train)

# 4. Generate predictions on unseen test data
y_pred = model.predict(X_test)

# 5. Model Parameters and Performance
beta_1 = model.coef_[0]
beta_0 = model.intercept_
mae = mean_absolute_error(y_test, y_pred)
rmse = np.sqrt(mean_squared_error(y_test, y_pred))
r2 = r2_score(y_test, y_pred)

print(f"Fitted Equation: waveHs1 = {beta_0:.3f} + ({beta_1:.3f} * waveHs2)")
print(f"Test MAE:       {mae:.3f} m")
print(f"Test RMSE:      {rmse:.3f} m")
print(f"Test R²:        {r2:.3f}")

# 6. Plot Observed vs. Predicted Time Series
fig, ax = plt.subplots(figsize=(12, 5))
ax.plot(
    test_times,
    y_test.values,
    label="Actual Station 1 (waveHs1)",
    color="#2ca02c",
    linewidth=1.2,

    y_pred,
    label=f"Predicted waveHs1 (Level 0 OLS)",

    linewidth=1.2,
)

ax.set_xlabel("Time")
ax.set_ylabel("Wave Height $H_s$ (m)")
ax.legend(loc="upper right")
ax.grid(True, linestyle=":", alpha=0.6)
