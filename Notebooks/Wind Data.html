import pandas as pd

def merge_wave_buoy_data(file_a: str, file_b: str, output_file: str):
    # Read both CSV files
    # Keeping station_id as string prevents leading zeros (like '098') from being dropped
    df_a = pd.read_csv(file_a, dtype={"station_id": str})
    df_b = pd.read_csv(file_b, dtype={"station_id": str})

    # Ensure time strings are consistently formatted/parsed before joining
    df_a["time"] = pd.to_datetime(df_a["time"])
    df_b["time"] = pd.to_datetime(df_b["time"])

    # Merge on 'time' using an inner join (records present in both files)
    merged_df = pd.merge(df_a, df_b, on="time", suffixes=("1", "2"), how="inner")

    # Arrange columns: place 'time' first, followed by station 1 fields, then station 2 fields
    non_time_cols = [c for c in df_a.columns if c != "time"]
    ordered_cols = (
        ["time"]
        + [f"{col}1" for col in non_time_cols]
        + [f"{col}2" for col in non_time_cols]
    )

    merged_df = merged_df[ordered_cols]

    # Save to CSV
    merged_df.to_csv(output_file, index=False)
    print(f"Successfully wrote {len(merged_df)} merged records to {output_file}")

    return merged_df

import matplotlib.dates as mdates
import matplotlib.pyplot as plt

# Ensure records are sorted chronologically

# Plot the difference
ax.plot(
    df["time"],
    df["diff_waveHs"],
    color="#1f77b4",
    linewidth=1.5,
    label="diff_waveHs (Station 225 - Station 098)",
)

# Reference line at zero
ax.axhline(0, color="gray", linestyle="--", linewidth=1, alpha=0.8)

# Color fill to highlight which station has higher waves
ax.fill_between(
    df["time"],
    df["diff_waveHs"],
    0,
    where=(df["diff_waveHs"] >= 0),
    color="crimson",
    alpha=0.15,
    label="Station 225 > Station 098",
)
ax.fill_between(
    df["time"],
    df["diff_waveHs"],
    0,
    where=(df["diff_waveHs"] < 0),
    color="navy",
    alpha=0.15,
    label="Station 098 > Station 225",
)

# Labels, formatting, and grid

ax.set_xlabel("Time", fontsize=11)
ax.set_ylabel("Difference in $H_s$ (m)", fontsize=11)

# Date formatting on the X-axis
